# 대용량 테이블 컬럼 타입 변경 무중단 처리 전략

- [CASE]
- 운영중인 대용량 테이블의 컬럼을 numeric(5,2) -> numeric(15,5) 로 변경할 경우
- [문제점]
- 물리적 스토리지 크기가 달라지기 때문에 MSSQL은 테이블의 모든 데이터 페이지를 재기록함,
- [해결 및 조치]
- Shadow Table + CDC 패턴
- 새로운 구조의 테이블을 만들어서 데이터를 복사하고, 복사 시간중에 발생한 변경분(Delta)를 CDC로 추적했다가
- 반영하고 , 향후에 이름만 바꾸는 방법.
- 이 방법을 반복하면 복사할 데이터의 양도 갈수록 작아지고, 복사 시간 중 발생한 변경분의 양도 적어질 것이기 때문에 복사 시간 중 발생한 변경분이 없을 때 까지 반복하여 반영하면 무중단으로 데이터를 이관할 수 있게된다.
- <기존 방식>
- Table 전체 X LOCK
- 전체 데이터 페이지 재기록
- 롤백 불가
- 작업 중 서비스 완전 중단 필요
- <CDC + Shadow table 방식>
- 서비스 영향 없이 복사 진행
- 중단 가능
- 다운타임 없음
- [CDC란 무엇인가?]
- Change Data Capture
- SQL server가 트랜잭션 로그를 읽어서 테이블의 변경 내역을 별도의 변경 테이블에 자동으로 기록하는 기능.
- LSN
- Log Sequence Number
- 트랜잭션 로그 내의 고유한 위치값.
- 특정 시점 이후에 발생한 변경만 가져오는 기준점으로 사용.
- Bulk Copy 시작 직전에 LSN을 기록해두면, 복사 중에 발생한 변경분만 정확히 추출 가능
```sql
-- 현재 로그의 최신 LSN 조회
SELECT sys.fn_cdc_get_max_lsn() AS current_lsn;
-- 예: 0x000FC10B0001EAC8004
```
- ​
- 작업 프로세스
- 1
- TEMP 테이블 생성
- 즉시 완료
- 새로운 컬럼 타입이 적용된 TEMP 테이블을 생성한다. 인덱스는 이 단계에서 만들지 않는다.
- 2
- CDC 활성화
- 즉시 완료
- 원본 테이블에 CDC를 활성화한다. 이 시점부터 모든 변경이 cdc 스키마에 기록되기 시작한다.
```sql
-- DB 레벨 (이미 활성화된 경우 skip)
IF NOT EXISTS (
SELECT 1 FROM sys.databases
WHERE name = DB_NAME() AND is_cdc_enabled = 1
)
BEGIN
EXEC sys.sp_cdc_enable_db;
END;
-- 테이블 레벨
EXEC sys.sp_cdc_enable_table
@source_schema = N'dbo',
@source_name   = N'대상테이블명',
@role_name     = NULL;
```
- ​
- 3
- LSN 기록 후 Bulk Copy 시작
- 수 시간 소요
- Pentaho 또는 INSERT...SELECT(배치)로 전체 데이터를 TEMP 테이블로 복사한다. 이 시간 동안 서비스는 정상 운영된다. CDC는 변경분을 조용히 수집한다.
```sql
SELECT
sys.fn_cdc_get_max_lsn() AS copy_start_lsn,
GETDATE()               AS recorded_at;
-- 결과 예: 0x000FC10B0001EAC80049
-- 이 값을 Delta Catch-up의 @from_lsn에 입력
```
- ​
- 4
- 인덱스 생성
- 서비스 영향 없음
- Bulk Copy 완료 후 TEMP 테이블에 인덱스를 생성한다. TEMP는 아직 서비스에 미투입 상태이므로 Lock 영향 없음.
```sql
CREATE INDEX IX_TEMP_대상테이블_01
ON dbo.TEMP_대상테이블 (컬럼1, 컬럼2)
WITH (SORT_IN_TEMPDB = ON, MAXDOP = 4) ON FG_데이터파일그룹;
-- SORT_IN_TEMPDB = ON : 인덱스 정렬을 tempdb에서 수행 → 데이터 파일그룹 I/O 분산
-- MAXDOP : Standard Edition에서도 사용 가능 (ONLINE 옵션만 Enterprise 전용)
-- ONLINE 옵션 불필요 : TEMP는 아직 서비스 미투입 상태이므로 Lock 영향 없음
```
- ​
- 5
- Delta Catch-up 반복
- 수 분
- 복사 중 쌓인 CDC 변경분을 TEMP에 반영한다. 건수가 수렴할 때까지 반복 실행한다.
```sql
DECLARE @from_lsn BINARY(10) = 0x/* copy_start_lsn 또는 직전 next_from_lsn */;
DECLARE @to_lsn   BINARY(10) = sys.fn_cdc_get_max_lsn();
SELECT
__$operation,
CASE __$operation
WHEN 1 THEN 'DELETE'
WHEN 2 THEN 'INSERT'
WHEN 3 THEN 'UPDATE(before)'
WHEN 4 THEN 'UPDATE(after)'
END AS op_type,
COUNT(*) AS cnt
FROM cdc.fn_cdc_get_all_changes_dbo_대상테이블(
@from_lsn, @to_lsn, N'all'
)
GROUP BY __$operation;
-- 1회차: 65만 건  → 반영 후 next_from_lsn 메모
-- 2회차:  8만 건  → 반영 후 next_from_lsn 메모
-- 3회차:  1만 건  → 수렴 판단, STEP 7 진행
```
- ​
- 6
- 최종 전환 (TABLOCKX + RENAME)
- 1분 이내
- 원본 테이블에 TABLOCKX를 걸어 Write를 차단하고, 마지막 Delta를 반영한 뒤 인덱스/PK/테이블 이름을 교체한다. 이 구간만 실질 다운타임이다.
- 7
- CDC 비활성화 및 구 테이블 삭제
- 검증 후 실행
- 데이터 검증이 완료된 후 CDC를 비활성화하고 구 원본 테이블(TEMP_...)을 삭제한다
- [작업 템플릿 예시]
```sql
- ================================================================
-- STEP 1. TEMP 테이블 생성 (인덱스 제외)
-- ================================================================
IF OBJECT_ID('dbo.TEMP_대상테이블', 'U') IS NOT NULL
DROP TABLE dbo.TEMP_대상테이블;
CREATE TABLE dbo.TEMP_대상테이블
(
컬럼1  CHAR(4) COLLATE Korean_Wansung_CI_AS NOT NULL,
-- ... 나머지 컬럼 원본과 동일하게 --
변경컬럼 NUMERIC(15,5) NULL,  -- ★ 변경 타입 적용
CONSTRAINT PK_TEMP_대상테이블
PRIMARY KEY CLUSTERED (PK컬럼들)
);
-- ================================================================
-- STEP 2. CDC 활성화
-- ================================================================
IF NOT EXISTS (SELECT 1 FROM sys.databases
WHERE name = DB_NAME() AND is_cdc_enabled = 1)
EXEC sys.sp_cdc_enable_db;
EXEC sys.sp_cdc_enable_table
@source_schema = N'dbo',
@source_name   = N'대상테이블',
@role_name     = NULL;
-- ================================================================
-- STEP 3. Pentaho 시작 직전 LSN 기록
-- !! Pentaho 실행 버튼 누르기 직전에 실행 !!
-- ================================================================
SELECT sys.fn_cdc_get_max_lsn() AS copy_start_lsn, GETDATE() AS recorded_at;
-- ================================================================
-- STEP 4. Pentaho Bulk Copy 실행 (수동)
-- ================================================================
-- 완료 후 STEP 5 진행
-- ================================================================
-- STEP 5. 인덱스 생성 (Bulk Copy 완료 후)
-- ================================================================
CREATE INDEX IX_TEMP_대상테이블_01
ON dbo.TEMP_대상테이블 (컬럼1, 컬럼2)
WITH (SORT_IN_TEMPDB = ON, MAXDOP = 4) ON FG_데이터파일그룹;
-- ================================================================
-- STEP 6. Delta Catch-up (건수 수렴까지 반복)
-- ================================================================
DECLARE @from_lsn BINARY(10) = 0x/* STEP 3 copy_start_lsn */;
DECLARE @to_lsn   BINARY(10) = sys.fn_cdc_get_max_lsn();
-- 건수 확인
SELECT __$operation, COUNT(*) AS cnt
FROM cdc.fn_cdc_get_all_changes_dbo_대상테이블(@from_lsn, @to_lsn, N'all')
GROUP BY __$operation;
-- INSERT 반영
INSERT INTO dbo.TEMP_대상테이블 (컬럼목록)
SELECT 컬럼목록
FROM cdc.fn_cdc_get_all_changes_dbo_대상테이블(@from_lsn, @to_lsn, N'all')
WHERE __$operation = 2;
-- UPDATE 반영 (after image)
UPDATE tgt SET /* 비PK 컬럼 = src.컬럼 */
FROM dbo.TEMP_대상테이블 tgt
INNER JOIN (
SELECT * FROM cdc.fn_cdc_get_all_changes_dbo_대상테이블(@from_lsn, @to_lsn, N'all')
WHERE __$operation = 4
) src ON /* PK 조인 조건 */;
-- DELETE 반영
DELETE tgt
FROM dbo.TEMP_대상테이블 tgt
INNER JOIN (
SELECT * FROM cdc.fn_cdc_get_all_changes_dbo_대상테이블(@from_lsn, @to_lsn, N'all')
WHERE __$operation = 1
) src ON /* PK 조인 조건 */;
SELECT @to_lsn AS next_from_lsn; -- 다음 반복 시 @from_lsn에 입력
-- ================================================================
-- STEP 7. 최종 전환 (TABLOCKX + RENAME)
-- ================================================================
BEGIN TRAN;
-- Write 차단
SELECT TOP 0 * FROM dbo.대상테이블 WITH (TABLOCKX, HOLDLOCK);
DECLARE @final_from_lsn BINARY(10) = 0x/* 마지막 next_from_lsn */;
DECLARE @final_to_lsn   BINARY(10) = sys.fn_cdc_get_max_lsn();
-- 마지막 Delta 반영 (INSERT/UPDATE/DELETE 동일 구조)
-- ... (STEP 6과 동일 로직)
-- 인덱스 RENAME (테이블 RENAME 이전에 수행)
EXEC sp_rename 'dbo.대상테이블.IX_SA_대상테이블_01',    'IX_WORK_대상테이블_01', 'INDEX';
EXEC sp_rename 'dbo.TEMP_대상테이블.IX_TEMP_대상테이블_01', 'IX_SA_대상테이블_01',   'INDEX';
EXEC sp_rename 'dbo.대상테이블.IX_WORK_대상테이블_01',    'IX_TEMP_대상테이블_01', 'INDEX';
-- PK RENAME (테이블 RENAME 이전에 수행)
EXEC sp_rename 'PK_대상테이블',      'PK_WORK_대상테이블', 'OBJECT';
EXEC sp_rename 'PK_TEMP_대상테이블', 'PK_대상테이블',      'OBJECT';
EXEC sp_rename 'PK_WORK_대상테이블', 'PK_TEMP_대상테이블', 'OBJECT';
-- 테이블 RENAME (마지막)
EXEC sp_rename 'dbo.대상테이블',      'WORK_대상테이블';
EXEC sp_rename 'dbo.TEMP_대상테이블',  '대상테이블';
EXEC sp_rename 'dbo.WORK_대상테이블',  'TEMP_대상테이블';
COMMIT;
-- ================================================================
-- STEP 8. 데이터 검증
-- ================================================================
SELECT 'SA(신규)'    AS tbl, COUNT(*) AS cnt FROM dbo.대상테이블
UNION ALL
SELECT 'TEMP(구원본)', COUNT(*) FROM dbo.TEMP_대상테이블;
-- ================================================================
-- STEP 9. CDC 비활성화 및 정리
-- ================================================================
EXEC sys.sp_cdc_disable_table
@source_schema    = N'dbo',
@source_name      = N'TEMP_대상테이블',       -- 구 원본이 이동한 이름
@capture_instance = N'dbo_대상테이블';        -- capture instance는 최초 이름
-- DROP TABLE dbo.TEMP_대상테이블;  -- 검증 완료 후 주석 해제
```
